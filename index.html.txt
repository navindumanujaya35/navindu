        // Helper: Check if vehicle is due for service (within 14 days or within 500km)
        function getServiceStatus(vehicle) {
            const today = new Date();
            today.setHours(0,0,0,0);
            const dueDate = new Date(vehicle.nextServiceDate);
            const diffDays = Math.ceil((dueDate - today) / (1000 * 60 * 60 * 24));
            const kmRemaining = vehicle.nextServiceKm - vehicle.mileage;

            if (diffDays < 0 || kmRemaining <= 0) {
                return { status: 'overdue', label: 'Overdue', badgeClass: 'bg-red-100 text-red-700 border-red-200', diffDays, kmRemaining };
            } else if (diffDays <= 14 || kmRemaining <= 1000) {
                return { status: 'due_soon', label: 'Due Soon', badgeClass: 'bg-amber-100 text-amber-800 border-amber-200', diffDays, kmRemaining };
            }
            return { status: 'ok', label: 'Scheduled', badgeClass: 'bg-emerald-50 text-emerald-700 border-emerald-200', diffDays, kmRemaining };
        }

        function openVehicleModal(vehicleId = null) {
            const form = document.getElementById('vehicle-form');
            form.reset();
            if (vehicleId) {
                const v = appData.vehicles.find(item => item.id === vehicleId);
                document.getElementById('vehicle-modal-title').textContent = 'Edit Vehicle Details';
                document.getElementById('v-id').value = v.id;
                document.getElementById('v-plate').value = v.plate;
                document.getElementById('v-model').value = v.model;
                document.getElementById('v-owner').value = v.owner;
                document.getElementById('v-phone').value = v.phone;
                document.getElementById('v-mileage').value = v.mileage;
                document.getElementById('v-next-km').value = v.nextServiceKm;
                document.getElementById('v-next-date').value = v.nextServiceDate;
            } else {
                document.getElementById('vehicle-modal-title').textContent = 'Register New Vehicle';
                document.getElementById('v-id').value = '';
                document.getElementById('v-next-date').value = getRelativeDate(90);
            }
            document.getElementById('vehicle-modal').classList.remove('hidden');
        }

        function saveVehicle(e) {
            e.preventDefault();
            const id = document.getElementById('v-id').value;
            const vehicleObj = {
                id: id || 'v_' + Date.now(),
                plate: document.getElementById('v-plate').value.toUpperCase().trim(),
                model: document.getElementById('v-model').value.trim(),
                owner: document.getElementById('v-owner').value.trim(),
                phone: document.getElementById('v-phone').value.trim(),
                mileage: Number(document.getElementById('v-mileage').value),
                nextServiceKm: Number(document.getElementById('v-next-km').value),
                nextServiceDate: document.getElementById('v-next-date').value
            };

            if (id) {
                const idx = appData.vehicles.findIndex(v => v.id === id);
                appData.vehicles[idx] = vehicleObj;
                showToast('Vehicle details updated!');
            } else {
                appData.vehicles.unshift(vehicleObj);
                showToast('New vehicle registered!');
            }
            saveData();
            closeModal('vehicle-modal');
        }

        function openHistoryModal(vehicleId) {
            const v = appData.vehicles.find(item => item.id === vehicleId);
            const vJobs = appData.jobs.filter(j => j.vehicleId === vehicleId);

            document.getElementById('history-modal-title').textContent = `${v.plate} — Service Log`;
            document.getElementById('history-modal-sub').textContent = `${v.model} • Owner: ${v.owner} (${v.phone})`;

            const body = document.getElementById('history-modal-body');
            if (vJobs.length === 0) {
                body.innerHTML = `<div class="text-center py-8 text-slate-400 text-sm">No completed service records found for this vehicle yet.</div>`;
            } else {
                body.innerHTML = vJobs.map(j => `
                    <div class="p-4 rounded-xl border border-slate-200 bg-slate-50/70 space-y-2">
                        <div class="flex items-center justify-between">
                            <span class="font-mono text-xs font-bold text-slate-700 bg-white px-2 py-0.5 rounded border border-slate-200">${j.id} • ${j.date}</span>
                            <span class="font-bold text-sm text-emerald-700">${formatLKR(j.total)}</span>
                        </div>
                        <p class="text-sm font-semibold text-slate-800">${j.notes}</p>
                        <p class="text-xs text-slate-500">Mileage at service: <strong>${Number(j.mileage).toLocaleString()} km</strong></p>
                        ${j.parts.length > 0 ? `
                            <div class="text-xs text-slate-600 pt-1 border-t border-slate-200/60">
                                <span class="font-semibold">Parts Used:</span> ${j.parts.map(p => `${p.name} (x${p.qty})`).join(', ')}
                            </div>
                        ` : ''}
                    </div>
                `).join('');
            }
            document.getElementById('history-modal').classList.remove('hidden');
        }

        function renderInventory() {
            const query = document.getElementById('part-search').value.toLowerCase().trim();
            const filter = document.getElementById('part-filter').value;
            const tbody = document.getElementById('inventory-table-body');

            const filtered = appData.parts.filter(p => {
                const matchesSearch = p.code.toLowerCase().includes(query) || p.name.toLowerCase().includes(query) || p.category.toLowerCase().includes(query);
                const matchesFilter = filter === 'all' || (filter === 'low' && p.stock <= p.minStock);
                return matchesSearch && matchesFilter;
            });

            if (filtered.length === 0) {
                tbody.innerHTML = `<tr><td colspan="5" class="py-8 text-center text-slate-400">No spare parts found.</td></tr>`;
                return;
            }

            tbody.innerHTML = filtered.map(p => {
                const isLow = p.stock <= p.minStock;
                return `
                    <tr class="hover:bg-slate-50/80 transition">
                        <td class="py-3.5 px-4 font-mono font-semibold text-xs text-slate-700">${p.code}</td>
                        <td class="py-3.5 px-4">
                            <p class="font-semibold text-slate-900">${p.name}</p>
                            <span class="text-xs text-slate-400">${p.category}</span>
                        </td>
                        <td class="py-3.5 px-4 text-right font-mono font-medium text-slate-800">${formatLKR(p.price)}</td>
                        <td class="py-3.5 px-4">
                            <div class="flex items-center justify-center gap-2">
                                <button onclick="quickAdjustStock('${p.id}', -1)" class="w-7 h-7 rounded-lg bg-slate-100 hover:bg-slate-200 text-slate-700 font-bold flex items-center justify-center transition">-</button>
                                <span class="w-10 text-center font-bold ${isLow ? 'text-red-600' : 'text-slate-800'}">${p.stock}</span>
                                <button onclick="quickAdjustStock('${p.id}', 1)" class="w-7 h-7 rounded-lg bg-slate-100 hover:bg-slate-200 text-slate-700 font-bold flex items-center justify-center transition">+</button>
                            </div>
                            ${isLow ? `<div class="text-center mt-1"><span class="text-[10px] font-bold uppercase px-2 py-0.5 rounded-full bg-red-100 text-red-700">Low Stock</span></div>` : ''}
                        </td>
                        <td class="py-3.5 px-4 text-right">
                            <button onclick="openPartModal('${p.id}')" class="p-2 text-slate-500 hover:text-slate-900 hover:bg-slate-100 rounded-lg transition" title="Edit Part">
                                <i class="ph-bold ph-pencil-simple"></i>
                            </button>
                        </td>
                    </tr>
                `;
            }).join('');
        }

        function quickAdjustStock(partId, delta) {
            const part = appData.parts.find(p => p.id === partId);
            if (!part) return;
            if (part.stock + delta < 0) {
                showToast('Stock cannot be negative!', 'error');
                return;
            }
            part.stock += delta;
            saveData();
        }
